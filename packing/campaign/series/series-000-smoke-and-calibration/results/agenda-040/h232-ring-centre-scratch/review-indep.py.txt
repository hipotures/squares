# my own core construction: half-diagonal vectors
def core(p):
    t,cx,cy,w,s=(F(v) for v in p)
    c=(1-t*t)/(1+t*t); sn=2*t/(1+t*t); h=s/2
    u=(c*h,sn*h); v=(-sn*h,c*h)
    V=[(cx+u[0]+v[0],cy+u[1]+v[1]),(cx-u[0]+v[0],cy-u[1]+v[1]),(cx-u[0]-v[0],cy-u[1]-v[1]),(cx+u[0]-v[0],cy+u[1]-v[1])]
    return dict(t=t,c=(cx,cy),w=w,V=V,cos=c,sin=sn)

# containment via square-frame coordinates (different method from the lane's cross-product sign test)
def contains(c,q):
    dx=q[0]-c["c"][0]; dy=q[1]-c["c"][1]
    u=dx*c["cos"]+dy*c["sin"]; v=-dx*c["sin"]+dy*c["cos"]

# alternative exact method: separating-axis test between core and the open box via strict inequalities on the 6 candidate axes
def meets_open_box(c,j):
    # core K (closed convex), box O open. K meets O iff no separating axis with K on one closed side and O on the other closed side
    # (for convex compact K and open convex O disjoint iff a line separates weakly with O open). Test the 4 edge normals of each.
    bx=[(F(1,2),F(1,2)),(F(1),F(1,2)),(F(1),F(1)),(F(1,2),F(1))]
    box=[ (lambda p:( (p[0] if corners[j][0]==0 else L-p[0]), (p[1] if corners[j][1]==0 else L-p[1]) ))(p) for p in bx]
    K=c["V"]
    axes=[(F(1),F(0)),(F(0),F(1)),(c["cos"],c["sin"]),(-c["sin"],c["cos"])]
    for a in axes:
        pk=[a[0]*x+a[1]*y for x,y in K]; pb=[a[0]*x+a[1]*y for x,y in box]
        if max(pk)<=min(pb) or max(pb)<=min(pk): return False   # weak separation suffices since box is open
    return True
