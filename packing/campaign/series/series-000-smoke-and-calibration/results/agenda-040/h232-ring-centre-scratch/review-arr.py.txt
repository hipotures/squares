def cross(P,Q,R,S):
    d1=(Q[0]-P[0],Q[1]-P[1]); d2=(S[0]-R[0],S[1]-R[1]); den=d1[0]*d2[1]-d1[1]*d2[0]

    rx=R[0]-P[0]; ry=R[1]-P[1]; t=(rx*d2[1]-ry*d2[0])/den; u=(rx*d1[1]-ry*d1[0])/den
    if 0<=t<=1 and 0<=u<=1: return (P[0]+t*d1[0],P[1]+t*d1[1])
