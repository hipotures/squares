def core(p):
    t, cx, cy, w, s = (F(v) for v in p)
    c = (1-t*t)/(1+t*t); sn = 2*t/(1+t*t)
    h = s/2
    e1 = (c*h, sn*h); e2 = (-sn*h, c*h)
    V = [(cx+e1[0]+e2[0], cy+e1[1]+e2[1]), (cx-e1[0]+e2[0], cy-e1[1]+e2[1]),
         (cx-e1[0]-e2[0], cy-e1[1]-e2[1]), (cx+e1[0]-e2[0], cy+e1[1]-e2[1])]
    return dict(t=t, cx=cx, cy=cy, w=w, s=s, cos=c, sin=sn, V=V)

C = [core(p) for p in pl]

def clip_poly(poly, hp):  # hp: (a,b,c) keep a*x+b*y<=c
    a,b,c = hp; out=[]
    m=len(poly)
    for i in range(m):
        P=poly[i]; Q=poly[(i+1)%m]
        fp=a*P[0]+b*P[1]-c; fq=a*Q[0]+b*Q[1]-c
        if fp<=0: out.append(P)
        if (fp<0<fq) or (fq<0<fp):
            tt=fp/(fp-fq); out.append((P[0]+tt*(Q[0]-P[0]), P[1]+tt*(Q[1]-P[1])))
    return out
def area(poly):
    if len(poly)<3: return F(0)
    return abs(sum(poly[i][0]*poly[(i+1)%len(poly)][1]-poly[(i+1)%len(poly)][0]*poly[i][1] for i in range(len(poly))))/2
def meets_open_box(V, x0,x1,y0,y1):
    p=V
    for hp in [(-1,0,-x0),(1,0,x1),(0,-1,-y0),(0,1,y1)]:
        p=clip_poly(p,hp)
        if not p: return False
    return area(p)>0
def contains_point(V, q):  # closed convex polygon (ccw or cw) contains q
    sgn=0
    for i in range(4):
        P=V[i]; Q=V[(i+1)%4]
        cr=(Q[0]-P[0])*(q[1]-P[1])-(Q[1]-P[1])*(q[0]-P[0])
        if cr==0: continue
        s=1 if cr>0 else -1
        if sgn==0: sgn=s
        elif s!=sgn: return False
    return True

half=F(1,2); one=F(1)

def box_at(j):
    x,y=corners[j]

    ys=(half,one) if y==0 else (L-one,L-half)
    return xs[0],xs[1],ys[0],ys[1]
def penetration(V,j):
    x,y=corners[j]
    return min((v[0] if x==0 else L-v[0])+(v[1] if y==0 else L-v[1]) for v in V)

def polys_overlap(V1,V2):
    p=V1
    for i in range(4):
        P=V2[i]; Q=V2[(i+1)%4]
        # half-plane of V2's edge containing V2 (V2 orientation)
        # determine orientation
        pass
    # generic: intersect V1 with V2 via clipping by each edge of V2 with proper orientation
    o = sum(V2[i][0]*V2[(i+1)%4][1]-V2[(i+1)%4][0]*V2[i][1] for i in range(4))
    for i in range(4):
        P=V2[i]; Q=V2[(i+1)%4]
        a=-(Q[1]-P[1]); b=(Q[0]-P[0])  # normal
        if o<0: a,b=-a,-b

        p=clip_poly(p,(-a,-b,-(a*P[0]+b*P[1])))
        if not p: return False
    return area(p)>0
