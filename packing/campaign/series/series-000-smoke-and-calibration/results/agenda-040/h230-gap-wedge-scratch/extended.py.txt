def sqdist(P, Q):
    """Exact squared distance between convex polygons (0 if closed sets meet)."""
    if polys_intersect(P, Q, strict=False): return F(0)
    best = None
    for A_, B_ in ((P, Q), (Q, P)):
        for (px, py) in A_:
            m = len(B_)
            for i in range(m):
                ax, ay = B_[i]; bx, by = B_[(i+1) % m]
                vx, vy = bx - ax, by - ay; l2 = vx*vx + vy*vy
                t = ((px - ax)*vx + (py - ay)*vy) / l2; t = max(F(0), min(F(1), t))
                dx, dy = px - (ax + t*vx), py - (ay + t*vy); d2 = dx*dx + dy*dy
                best = d2 if best is None else min(best, d2)
    return best

def A_num(th, g):
